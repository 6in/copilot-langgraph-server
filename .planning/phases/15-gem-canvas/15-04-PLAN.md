function parseJobResult(raw: string): { text: string; canvas: CanvasResult | null } {
  try {
    const parsed = JSON.parse(raw);
    if (parsed && parsed.type === 'canvas') {
      return { text: raw, canvas: parsed as CanvasResult };
    }
  } catch {
    // plain text
  }
  return { text: raw, canvas: null };
}
