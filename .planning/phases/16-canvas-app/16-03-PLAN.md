    const handleOpenCanvas = () => { setCurrentScreen('canvas'); };

    const handleOpenCanvasChat = async (initialThreadId?: string) => {
      // canvasGemId が未取得の場合は取得してから遷移
      let gemId = canvasGemId;
      if (!gemId) {
        try {
          const result = await getCanvasGemId();
          gemId = result.gem_id;
          setCanvasGemId(gemId);
        } catch {
          // 取得失敗でも遷移は許可（CanvasChatApp 側でエラーハンドリング）
        }
      }
      setActiveCanvasAppId(initialThreadId ?? null);
      setCurrentScreen('canvaschat');
    };

    const handleBackFromCanvasChat = () => {
      setCurrentScreen('canvas');
    };

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| App.tsx → getCanvasGemId() | JWT cookie 付きリクエスト — 認証済みユーザーのみ canvas gem_id を取得可能 |
| CanvasChatApp canvasGemId prop | サーバーから取得した UUID をそのまま渡す — ユーザーが書き換えできない |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-16-09 | Spoofing | canvasGemId state の初期化 | mitigate | canvasGemId が null のとき canvaschat 画面を表示しない（`canvasGemId &&` 条件）。CanvasChatApp に null が渡ることを TypeScript 型レベルで防ぐ（prop が string 型）|
| T-16-10 | Information Disclosure | handleOpenCanvasChat のエラーハンドリング | accept | getCanvasGemId() 失敗時は canvasGemId が null のまま — canvaschat が表示されないのでアプリが安全に degraded 状態になる |
| T-16-11 | Elevation of Privilege | activeCanvasAppId（initialThreadId） | accept | thread_id は canvas_apps.thread_id から取得した UUID。バックエンドで所有権チェック済み（canvas.py の github_login フィルタ） |
</threat_model>
