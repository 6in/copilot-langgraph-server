```python
"""Hosted Canvas Apps — dynamic hosting shell (Phase 19).

GET /apps/{app_id} — DB から HTML を取得し、iframe srcdoc に埋め込んだ
フルスクリーンシェル HTML を返す。認証不要（D-08）。

DB に HTML が存在しない場合は 404 を返す（D-12）。
"""
import psycopg
from fastapi import APIRouter, HTTPException, Request
from fastapi.responses import HTMLResponse
from psycopg.rows import dict_row

router = APIRouter(tags=["hosted-apps"])

_SHELL_TEMPLATE = """<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>{app_name}</title>
<style>
  * {{ margin: 0; padding: 0; box-sizing: border-box; }}
  html, body {{ height: 100%; overflow: hidden; }}
  iframe {{ width: 100%; height: 100%; border: none; display: block; }}
</style>
</head>
<body>
<iframe
  sandbox="allow-scripts allow-forms"
  srcdoc="{srcdoc_escaped}"
></iframe>
<script type="module" src="/js/iframe-rpc.js"></script>
</body>
</html>"""


@router.get("/apps/{app_id}", response_class=HTMLResponse)
async def serve_hosted_app(app_id: str, request: Request) -> HTMLResponse:
    """Serve a deployed Canvas app as a standalone page (D-01, D-08).

    Fetches HTML from DB (no auth required), embeds in srcdoc iframe shell.
    Returns 404 if app not found (D-12).
    """
    db_uri = request.app.state.db_uri

    async with await psycopg.AsyncConnection.connect(db_uri, row_factory=dict_row) as conn:
        async with conn.cursor() as cur:
            await cur.execute(
                "SELECT name, html FROM canvas_apps WHERE app_id = %s::uuid",
                (app_id,),
            )
            row = await cur.fetchone()

    if row is None or not row["html"]:
        raise HTTPException(status_code=404, detail="Canvas app not found")

    # srcdoc 属性に埋め込むため HTML 内の " と & をエスケープ
    srcdoc_escaped = row["html"].replace("&", "&amp;").replace('"', "&quot;")
    app_name = row["name"] or "Canvas App"

    html = _SHELL_TEMPLATE.format(
        app_name=app_name,
        srcdoc_escaped=srcdoc_escaped,
    )
    return HTMLResponse(content=html)
```

**注意事項:**
- `github_login` フィルターなし — app_id のみで検索（D-05 調整版、D-08）
- `sandbox="allow-scripts allow-forms"` のみ。`allow-same-origin` は含めない（D-06）
- `<script type="module" src="/js/iframe-rpc.js">` でブリッジ読み込み（D-13）
- DB に HTML がない場合（html が None または空文字）は 404（D-12）
- srcdoc エスケープ: `"` → `&quot;`、`&` → `&amp;`（HTML 属性値として安全に埋め込むため）
- `{` と `}` を Python format 文字列テンプレートで使うため CSS の `{{ }}` に注意

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| 外部→GET /apps/{app_id} | 認証なし。app_id を知る任意ユーザーがアクセス可能 |
| iframe srcdoc→ブラウザ | Canvas アプリ HTML が sandboxed iframe で実行される |
| iframe→POST /api/iframe-rpc | D-07 により JWT 不要。arq へ enqueue される |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-19-01 | Information Disclosure | GET /apps/{app_id} | accept | D-08 決定: URL を知る者は誰でもアクセス可能。社内ツール・200人規模なので許容。将来フェーズで JWT チェック追加予定 |
| T-19-02 | Tampering | srcdoc 注入（HTML エスケープ） | mitigate | DB の HTML を srcdoc 属性値に埋め込む前に `"` → `&quot;`、`&` → `&amp;` でエスケープ。これにより属性値の早期終端を防ぐ |
| T-19-03 | Elevation of Privilege | iframe sandbox の破壊 | mitigate | `sandbox="allow-scripts allow-forms"` のみ。`allow-same-origin` を除外することで iframe 内 JS が parent DOM にアクセス不可（Phase 15 D で確立済み） |
| T-19-04 | Spoofing | POST /api/iframe-rpc JWT 削除後の不正利用 | accept | D-07 の暫定決定。iframe-rpc はアプリロジック専用（SELECT + AI ワンショット）でありデータ破壊操作なし。INSERT/UPDATE/DELETE は D-09（Phase 18）で拒否済み |
| T-19-05 | Information Disclosure | DB app_id 列挙 | mitigate | UUIDv4 の app_id は予測不可能（128bit）。総当たりは非現実的 |
| T-19-06 | Denial of Service | 大量 GET /apps リクエスト | accept | 社内 200 名規模。レートリミットは現フェーズ対象外 |
</threat_model>
