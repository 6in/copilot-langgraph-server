    ```python
    def capture_trace_logs(caplog):
        """Capture 'trace' logger JSON lines as parsed span dicts.

            return spans
    ```

    ```python
    from __future__ import annotations
    """TRACE_* env var ヘルパー。Phase 31 D-11 per-tool redact 禁止方針のため全ツール一律。"""
    import os

    _DEFAULT_ARGS_MAX = 500
    _DEFAULT_RESULT_MAX = 1000


    def get_args_max_chars() -> int:
        """TRACE_ARGS_MAX_CHARS env var を int で返す (default 500)。不正値は default。"""
        raw = os.environ.get("TRACE_ARGS_MAX_CHARS")
        if not raw:
            return _DEFAULT_ARGS_MAX
        try:
            v = int(raw)
            return v if v >= 0 else _DEFAULT_ARGS_MAX
        except ValueError:
            return _DEFAULT_ARGS_MAX


    def get_result_max_chars() -> int:
        """TRACE_RESULT_MAX_CHARS env var を int で返す (default 1000)。不正値は default。"""
        raw = os.environ.get("TRACE_RESULT_MAX_CHARS")
        if not raw:
            return _DEFAULT_RESULT_MAX
        try:
            v = int(raw)
            return v if v >= 0 else _DEFAULT_RESULT_MAX
        except ValueError:
            return _DEFAULT_RESULT_MAX
    ```
