    # 送信後 30 秒待ち
    sleep 30
    
    # trace_id を抽出
    docker compose logs --since 2m api worker 2>&1 | \
      python3 scripts/trace_query.py --app chat --operation request --format ndjson | \
      tail -5
    
    # 期待: handler=langgraph, app_id=chat, operation_name=request の span が 1 件以上
    ```

    - [ ] request span に `handler=langgraph` attribute がある
    - [ ] `trace_id` が UUID4 形式
    - [ ] `status_code=OK`

    sleep 30
    
    # tool_call span を抽出

      python3 scripts/trace_query.py --tool web_search --format ndjson | \
      tail -1 | jq -r .trace_id)
    echo "web_search trace_id: $TRACE_ID_WS"
    
    # 同 trace_id の全 span を tree 表示
    docker compose logs --since 2m api worker 2>&1 | \
      python3 scripts/trace_query.py --trace-id "$TRACE_ID_WS" --format tree
    ```

    - [ ] tree 出力に request → routing → sub_agent → tool_call(web_search) の 4 層が現れる

    - [ ] parent_span_id で親子関係が正しくつながっている (tree で ├── └── が正しく配置)
