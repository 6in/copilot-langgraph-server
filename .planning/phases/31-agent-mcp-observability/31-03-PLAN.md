    ```python
    from __future__ import annotations
    """LangChain BaseTool wrapper で tool_call span を自動 emit する (Phase 31 D-10..D-12)."""

    import json
    import logging
    from typing import Any, Callable

    from langchain_core.tools import BaseTool
    from pydantic import ConfigDict, PrivateAttr

    from app.observability.trace import trace_span, _current_trace_id, _current_common_attrs
    from app.observability.config import get_args_max_chars, get_result_max_chars

    logger = logging.getLogger(__name__)


    class TracedTool(BaseTool):
        """BaseTool の透過ラッパー。実行前後に tool_call span を emit する。

        Args:
            wrapped: ラップ対象 BaseTool
            privileged_tool_names: sandbox_exposed=false のツール名 frozenset。
                wrapped.name がここに含まれば attributes.privileged=True。
            common_attrs_provider: 任意。呼び出しごとに attributes に merge する共通属性を返す callable。
                **W2 反映: 指定がない場合は `_current_common_attrs` ContextVar から自動取得する**
                (sub_agent span が set した user_id / app_id / agent_name / model_name)。
                指定がある場合は ContextVar より provider を優先する。
        """
        model_config = ConfigDict(arbitrary_types_allowed=True)

        _wrapped: BaseTool = PrivateAttr()
        _privileged_names: frozenset[str] = PrivateAttr()
        _common_attrs_provider: Callable[[], dict[str, Any]] = PrivateAttr()

        def __init__(
            self,
            wrapped: BaseTool,
            privileged_tool_names: frozenset[str] = frozenset(),
            common_attrs_provider: Callable[[], dict[str, Any]] | None = None,
            **data,
        ):
            super().__init__(
                name=wrapped.name,
                description=wrapped.description,
                args_schema=getattr(wrapped, "args_schema", None),
                **data,
            )
            object.__setattr__(self, "_wrapped", wrapped)
            object.__setattr__(self, "_privileged_names", privileged_tool_names)
            # W2 反映: None を許容。_arun で None なら ContextVar 経由で解決
            object.__setattr__(self, "_common_attrs_provider", common_attrs_provider)

        def _run(self, *args, **kwargs):
            # Phase 31 scope: async only (全経路が async tool.ainvoke を使う)
            raise NotImplementedError("TracedTool supports async execution only. Use ainvoke().")

        async def _arun(self, **kwargs) -> Any:
            trace_id = _current_trace_id.get() or "unknown"
            args_max = get_args_max_chars()
            result_max = get_result_max_chars()

            args_json = json.dumps(kwargs, ensure_ascii=False, default=str)
            # W2 反映: common_attrs_provider が None なら ContextVar (_current_common_attrs) を参照
            if self._common_attrs_provider is None:
                common = dict(_current_common_attrs.get() or {})
            else:
                common = dict(self._common_attrs_provider())

            attributes = {
                **common,
                "tool_name": self._wrapped.name,
                "args_bytes": len(args_json.encode("utf-8")),
                "args_prefix": args_json[:args_max],
                "privileged": self._wrapped.name in self._privileged_names,
            }

            async with trace_span("tool_call", trace_id=trace_id, attributes=attributes) as span:
                result = await self._wrapped.ainvoke(kwargs)
                result_json = json.dumps(result, ensure_ascii=False, default=str)
                span.set_attribute("result_bytes", len(result_json.encode("utf-8")))
                span.set_attribute("result_prefix", result_json[:result_max])

                if isinstance(result, dict) and "error" in result:
                    span.set_attribute("success", False)
                    span.set_status("ERROR", str(result["error"])[:200])
                else:
                    span.set_attribute("success", True)

                return result
    ```

    重要な実装決定:
    - **trace_id は `_current_trace_id.get()`** から引く（ContextVar 経由、RESEARCH §3.1 推奨）。caller (Plan 04/05) が `async with trace_span("sub_agent", trace_id=...)` で外側を包むと自動で伝搬。外側がない場合は "unknown" string で emit（iframe_rpc の legacy 経路などへの defensive default）
    - **bytes** は `len(json_str.encode("utf-8"))` で UTF-8 bytes 数を使う（日本語対応）
    - **privileged** は frozenset 判定で boolean。D-12: アラートは Phase 31 スコープ外、attribute 記録のみ
    - **_run は NotImplementedError** — langchain_core BaseTool は sync も要求するが、Phase 31 の全 ツール経路は async のみなので明示拒否
