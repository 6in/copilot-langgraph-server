```typescript
export function useAgents(): UseAgentsReturn {
  const [agents, setAgents] = useState<AgentInfo[]>([]);
  const [selectedAgents, setSelectedAgents] = useState<Set<string>>(new Set());
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    setIsLoading(true);
    getAgents()
      .then((data) => {
        if (cancelled) return;
        setAgents(data);
        setSelectedAgents(new Set(data.map((a) => a.name)));
      })
      // ...
  }, []);

  const toggleAgent = useCallback((name: string) => {
    setSelectedAgents((prev) => {
      const next = new Set(prev);
      if (next.has(name)) {
        if (next.size <= 1) return prev;  // 最後の1つは削除不可
        next.delete(name);
      } else {
        next.add(name);
      }
      return next;
    });
  }, []);

  return { agents, selectedAgents: Array.from(selectedAgents), toggleAgent, isLoading };
}
```

**frontend/src/hooks/useGems.ts の新規作成（D-14）:**

`useAgents.ts` と対称的な実装。`GET /api/gems` から Gem 一覧を取得する。

重要な違い:
- Gem はデフォルト「全未選択」（agents は全選択デフォルト）— ユーザーが明示的に招待する設計
- toggleGem は「最後の1つを削除不可」制限を設けない（0個選択 = Gem 招待なし が有効）
- `getGems()` API クライアント関数を使用する（未実装なら `apiFetch('/api/gems')` で実装）

```typescript
// frontend/src/hooks/useGems.ts
import { useCallback, useEffect, useState } from 'react';
import { apiFetch } from '../api/client';
import type { GemInfo } from '../types';

interface UseGemsReturn {
  gems: GemInfo[];
  selectedGemIds: string[];
  toggleGem: (gemId: string) => void;
  isLoading: boolean;
}

export function useGems(): UseGemsReturn {
  const [gems, setGems] = useState<GemInfo[]>([]);
  const [selectedGemIds, setSelectedGemIds] = useState<Set<string>>(new Set());
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    setIsLoading(true);
    apiFetch('/api/gems')
      .then((data: GemInfo[]) => {
        if (cancelled) return;
        setGems(data);
        // デフォルト全未選択（ユーザーが明示的に招待する設計）
      })
      .catch(() => { /* silently fail */ })
      .finally(() => { if (!cancelled) setIsLoading(false); });
    return () => { cancelled = true; };
  }, []);

  const toggleGem = useCallback((gemId: string) => {
    setSelectedGemIds((prev) => {
      const next = new Set(prev);
      if (next.has(gemId)) {
        next.delete(gemId);  // 0個選択も有効（Gem 招待なしを意味する）
      } else {
        next.add(gemId);
      }
      return next;
    });
  }, []);

  return {
    gems,
    selectedGemIds: Array.from(selectedGemIds),
    toggleGem,
    isLoading,
  };
}
```

```tsx
// frontend/src/components/GemSelector.tsx
import type { GemInfo } from '../types';

interface GemChipProps {
  gem: GemInfo;
  selected: boolean;
  onToggle: (gemId: string) => void;
}

function GemChip({ gem, selected, onToggle }: GemChipProps) {
  return (
    <button
      onClick={() => onToggle(gem.gem_id)}
      title={gem.description || gem.name}
      style={{
        padding: '4px 12px',
        borderRadius: '16px',
        border: `1px solid ${selected ? '#28a745' : '#d1dbe3'}`,
        background: selected ? '#28a745' : 'transparent',
        color: selected ? '#fff' : '#555',
        fontSize: '0.82rem',
        fontWeight: selected ? 600 : 400,
        cursor: 'pointer',
        transition: 'all 0.15s',
        whiteSpace: 'nowrap',
        flexShrink: 0,
      }}
    >
      {gem.name}
    </button>
  );
}

interface GemSelectorProps {
  gems: GemInfo[];
  selectedGemIds: string[];
  onToggle: (gemId: string) => void;
  isLoading: boolean;
}

export function GemSelector({ gems, selectedGemIds, onToggle, isLoading }: GemSelectorProps) {
  if (!isLoading && gems.length === 0) return null;  // Gem が1つもない場合は行を表示しない

  const selectedSet = new Set(selectedGemIds);
  return (
    <div
      style={{
        display: 'flex',
        alignItems: 'center',
        gap: '6px',
        padding: '6px 12px',
        borderBottom: '1px solid #d1dbe3',
        background: '#f8f9fa',
        overflowX: 'auto',
        flexShrink: 0,
        minHeight: '38px',
      }}
    >
      <span
        style={{
          fontSize: '0.78rem',
          color: '#777',
          whiteSpace: 'nowrap',
          flexShrink: 0,
          marginRight: '4px',
        }}
      >
        Gems:
      </span>
      {isLoading ? (
        <span style={{ fontSize: '0.78rem', color: '#aaa' }}>Loading...</span>
      ) : (
        gems.map((gem) => (
          <GemChip
            key={gem.gem_id}
            gem={gem}
            selected={selectedSet.has(gem.gem_id)}
            onToggle={onToggle}
          />
        ))
      )}
    </div>
  );
}
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-16-04 | Spoofing | POST /api/chat gem_ids | mitigate | JWT 認証必須（get_github_token dependency）。未認証リクエストは 401 |
