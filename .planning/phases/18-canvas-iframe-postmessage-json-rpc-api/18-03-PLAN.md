       ```typescript
       const handleIframeMessage = useCallback(async (e: MessageEvent) => {
         // D-02: Origin validation
         // Deployed apps at /apps/{id}/ share window.location.origin.
         // srcDoc preview iframe has origin 'null' (string) — allow both.
         if (e.origin !== window.location.origin && e.origin !== 'null') return;

         // JSON-RPC format check
         const msg = e.data;
         if (!msg || typeof msg !== 'object' || msg.jsonrpc !== '2.0') return;

         const { id, method, params } = msg;
         if (!id || !method) return;

         try {
           // D-04: POST /api/iframe-rpc to enqueue job
           const { job_id } = await postIframeRpc(id, method, params);

           // SSE: wait for job completion via polling (simpler than EventSource for this use case)
           const result = await new Promise<string>((resolve, reject) => {
             const pollInterval = setInterval(async () => {
               try {
                 const job = await getJob(job_id);
                 if (job.status === 'done' && job.result !== undefined) {
                   clearInterval(pollInterval);
                   resolve(job.result);
                 }
               } catch {
                 clearInterval(pollInterval);
                 reject(new Error('Job polling failed'));
               }
             }, 500);

             // Timeout after 60 seconds
             setTimeout(() => {
               clearInterval(pollInterval);
               reject(new Error('RPC timeout'));
             }, 60000);
           });

           // Parse result and send back to iframe
           let parsed: unknown;
           try {
             parsed = JSON.parse(result);
           } catch {
             parsed = { result: false, error: 'Invalid response format' };
           }

           // D-05: Return with matching JSON-RPC id
           // targetOrigin '*' required for srcDoc iframe (null origin)
           iframeRef.current?.contentWindow?.postMessage(
             { jsonrpc: '2.0', id, ...((parsed && typeof parsed === 'object') ? parsed : { result: parsed }) },
             '*'
           );
         } catch (err) {
           // Send error back to iframe
           iframeRef.current?.contentWindow?.postMessage(
             { jsonrpc: '2.0', id, result: false, error: err instanceof Error ? err.message : 'Unknown error' },
             '*'
           );
         }
       }, []);
       ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| iframe JS → parent postMessage | 信頼されない iframe コードからの JSON-RPC リクエスト |
| parent → iframe postMessage | 処理結果の返信（targetOrigin='*'） |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-18-09 | Spoofing | postMessage origin | mitigate | e.origin を window.location.origin または 'null' (srcDoc) に限定。他の origin は無視 |
| T-18-10 | Spoofing | postMessage format | mitigate | msg.jsonrpc === '2.0' + id + method の存在チェック。不正形式は無視 |
| T-18-11 | Information Disclosure | targetOrigin '*' | accept | srcDoc iframe は null origin のため '*' が必須。iframe sandbox で allow-same-origin なしにより、iframe が parent DOM にアクセスすることは不可能。リスクは低い |
| T-18-12 | Denial of Service | 大量 postMessage | accept | 各リクエストは独立した job_id + SSE ポーリングで処理。arq ワーカーのスケジュールで自然に制限される。社内利用（200名）のため過剰対策不要 |
</threat_model>
